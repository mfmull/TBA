var addGeom = function(f) {
  var g = f.geometry();
  var area_km2  = g.area(1).divide(1e6);
  var perim_km  = g.perimeter(1).divide(1000);
  var compact   = ee.Number(4).multiply(Math.PI).multiply(area_km2)
                    .divide(perim_km.pow(2));         // Polsby–Popper
  // Oriented bounding rectangle elongation proxy
  var rect = g.bounds();
  var coords = ee.List(rect.coordinates().get(0));
  var xs = ee.List.sequence(0, coords.length().subtract(1)).map(function(i){return ee.Number(ee.List(coords.get(i)).get(0));});
  var ys = ee.List.sequence(0, coords.length().subtract(1)).map(function(i){return ee.Number(ee.List(coords.get(i)).get(1));});
  var w = ee.Number(xs.sort().get(-1)).subtract(ee.Number(xs.sort().get(0))).abs();
  var h = ee.Number(ys.sort().get(-1)).subtract(ee.Number(ys.sort().get(0))).abs();
  var elong = w.max(h).divide(w.min(h));
  var lat_c = ee.Number(g.centroid(1).coordinates().get(1));
  var lon_c = ee.Number(g.centroid(1).coordinates().get(0));

  return f.set({area_km2: area_km2, perim_km: perim_km, compact: compact, elong: elong, lat_c: lat_c,lon_c:lon_c});
};
var aqf_geom = aqf.map(addGeom);
